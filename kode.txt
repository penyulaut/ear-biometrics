def augment_training_images(images, labels, n_aug=2):
    augmented_images = []
    augmented_labels = []

    for img, label in zip(images, labels):
        # Gambar asli tetap digunakan
        augmented_images.append(img)
        augmented_labels.append(label)

        # Tambahkan dua variasi gambar
        for _ in range(n_aug):
            img_float = img.astype(np.float32) / 255.0

            img_aug = data_augmentation_fusion(
                tf.expand_dims(img_float, axis=0),
                training=True
            )[0]

            img_aug = tf.clip_by_value(img_aug, 0.0, 1.0)
            img_aug = (img_aug.numpy() * 255).round().astype(np.uint8)

            augmented_images.append(img_aug)
            augmented_labels.append(label)

    return (
        np.asarray(augmented_images, dtype=np.uint8),
        np.asarray(augmented_labels, dtype=np.int32)
    )

GLCM_DISTANCES = [1, 2]
GLCM_ANGLES = [
    0,
    np.pi / 4,
    np.pi / 2,
    3 * np.pi / 4
]

GLCM_PROPERTIES = [
    "contrast",
    "correlation",
    "energy",
    "homogeneity"
]


def extract_glcm_features(image_rgb):
    # Ubah RGB menjadi grayscale
    gray = np.asarray(
        Image.fromarray(image_rgb).convert("L")
    )

    # Kuantisasi ke 32 level keabuan: 0 sampai 31
    gray_quantized = (gray // 8).astype(np.uint8)

    glcm = graycomatrix(
        gray_quantized,
        distances=GLCM_DISTANCES,
        angles=GLCM_ANGLES,
        levels=32,
        symmetric=True,
        normed=True
    )

    features = []

    for prop in GLCM_PROPERTIES:
        values = graycoprops(glcm, prop)
        features.extend(values.flatten())

    return np.asarray(features, dtype=np.float32)


def extract_glcm_dataset(images):
    features = [
        extract_glcm_features(img)
        for img in images
    ]
    return np.asarray(features, dtype=np.float32)

def build_model(num_classes):
    image_input = layers.Input(
        shape=(IMG_SIZE, IMG_SIZE, 3),
        name="image_input"
    )

    backbone = MobileNetV2(
        weights="imagenet",
        include_top=False,
        input_shape=(IMG_SIZE, IMG_SIZE, 3)
    )
    backbone.trainable = False

    x = data_augmentation(image_input)
    x = backbone(x, training=False)
    x = layers.GlobalAveragePooling2D()(x)
    x = layers.Dropout(0.3)(x)
    x = layers.Dense(
        128,
        activation="relu",
        kernel_regularizer=regularizers.l2(1e-3)
    )(x)
    x = layers.BatchNormalization()(x)
    x = layers.Dropout(0.5)(x)

    output = layers.Dense(
        num_classes,
        activation="softmax",
        name="classification"
    )(x)

    model = Model(image_input, output)
    return model, backbone

def compile_model(model, learning_rate=1e-4):
    model.compile(
        optimizer=tf.keras.optimizers.Adam(
            learning_rate=learning_rate
        ),
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"]
    )


def get_callbacks(checkpoint_path):
    return [
        EarlyStopping(
            monitor="val_loss",
            patience=8,
            restore_best_weights=True,
            verbose=1
        ),
        ReduceLROnPlateau(
            monitor="val_loss",
            factor=0.2,
            patience=3,
            min_lr=1e-7,
            verbose=1
        ),
        ModelCheckpoint(
            checkpoint_path,
            monitor="val_loss",
            save_best_only=True,
            verbose=1
        )
    ]


def train_model(
    model,
    train_inputs,
    val_inputs,
    y_train,
    y_val,
    checkpoint_path,
    epochs=20,
    learning_rate=1e-3
):
    compile_model(model, learning_rate)

    history = model.fit(
        train_inputs,
        y_train,
        validation_data=(val_inputs, y_val),
        epochs=epochs,
        batch_size=BATCH_SIZE,
        class_weight=class_weights,
        callbacks=get_callbacks(checkpoint_path),
        verbose=1
    )

    return history

def plot_training_history(history, model_name):
    if history is None:
        print(f"Riwayat training {model_name} tidak tersedia.")
        return

    hist = history.history

    if "loss" not in hist or "val_loss" not in hist:
        print(f"Data loss untuk {model_name} tidak lengkap.")
        return

    epochs = range(1, len(hist["loss"]) + 1)

    fig, axes = plt.subplots(1, 2, figsize=(14, 5))

    # Grafik akurasi
    if "accuracy" in hist and "val_accuracy" in hist:
        axes[0].plot(
            epochs, hist["accuracy"],
            marker="o", label="Training"
        )
        axes[0].plot(
            epochs, hist["val_accuracy"],
            marker="o", label="Validation"
        )
        axes[0].set_title(f"{model_name} - Accuracy")
        axes[0].set_ylabel("Accuracy")
        axes[0].set_ylim(0, 1.05)
        axes[0].legend()
    else:
        axes[0].set_title(f"{model_name} - Accuracy tidak tersedia")

    axes[0].set_xlabel("Epoch")
    axes[0].grid(True, alpha=0.3)

    # Grafik loss
    axes[1].plot(
        epochs, hist["loss"],
        marker="o", label="Training"
    )
    axes[1].plot(
        epochs, hist["val_loss"],
        marker="o", label="Validation"
    )
    axes[1].set_title(f"{model_name} - Loss")
    axes[1].set_xlabel("Epoch")
    axes[1].set_ylabel("Loss")
    axes[1].grid(True, alpha=0.3)
    axes[1].legend()

    plt.tight_layout()
    plt.show()

def evaluate_model(model, inputs, y_true, model_name):
    inputs = np.asarray(inputs)
    y_true = np.asarray(y_true).astype(int)

    # Periksa jumlah fitur sebelum prediksi
    expected_shape = model.input_shape
    expected_features = expected_shape[-1]

    # Toleransi bentuk input gambar 4D untuk MobileNetV2
    if len(expected_shape) == 4:
        if inputs.ndim != 4:
            raise ValueError(
                f"{model_name}: input harus berupa array 4D, "
                f"tetapi bentuknya {inputs.shape}."
            )
    else:
        if inputs.ndim != 2:
            raise ValueError(
                f"{model_name}: input harus berupa array 2D, "
                f"tetapi bentuknya {inputs.shape}."
            )

        if inputs.shape[1] != expected_features:
            raise ValueError(
                f"{model_name}: jumlah fitur tidak sesuai. "
                f"Model mengharapkan {expected_features} fitur, "
                f"sedangkan data memiliki {inputs.shape[1]} fitur."
            )

    if len(inputs) != len(y_true):
        raise ValueError(
            f"{model_name}: jumlah data ({len(inputs)}) "
            f"tidak sama dengan jumlah label ({len(y_true)})."
        )

    probabilities = model.predict(inputs, verbose=0)
    y_pred = np.argmax(probabilities, axis=1)

    # Metrik evaluasi
    metrics = {
        "Model": model_name,
        "Accuracy": accuracy_score(y_true, y_pred),
        "Precision (Macro)": precision_score(
            y_true, y_pred,
            average="macro", zero_division=0
        ),
        "Recall (Macro)": recall_score(
            y_true, y_pred,
            average="macro", zero_division=0
        ),
        "F1-score (Macro)": f1_score(
            y_true, y_pred,
            average="macro", zero_division=0
        )
    }

    print(f"\n{'=' * 15} {model_name} {'=' * 15}")
    print(f"Bentuk input      : {inputs.shape}")
    print(f"Jumlah data test  : {len(y_true)}")
    print(f"Prediksi benar    : {(y_pred == y_true).sum()}")
    print(f"Prediksi salah    : {(y_pred != y_true).sum()}")
    print(f"Akurasi test      : {metrics['Accuracy']:.4f}")
    print(f"Precision Macro   : {metrics['Precision (Macro)']:.4f}")
    print(f"Recall Macro      : {metrics['Recall (Macro)']:.4f}")
    print(f"F1-score Macro    : {metrics['F1-score (Macro)']:.4f}")

    # Classification report
    print("\nClassification Report:")

    print(classification_report(
        y_true,
        y_pred,
        labels=np.arange(len(class_names)),
        target_names=class_names,
        zero_division=0,
        digits=4
    ))

    # Confusion matrix
    cm = confusion_matrix(
        y_true,
        y_pred,
        labels=np.arange(len(class_names))
    )

    row_sums = cm.sum(axis=1, keepdims=True)

    cm_normalized = np.divide(
        cm.astype(float),
        row_sums,
        out=np.zeros_like(cm, dtype=float),
        where=row_sums != 0
    )

    # Tampilkan angka aktual dan persentase per kelas
    annotations = np.empty(cm.shape, dtype=object)

    for i in range(cm.shape[0]):
        for j in range(cm.shape[1]):
            annotations[i, j] = (
                f"{cm[i, j]}\n"
                f"({cm_normalized[i, j] * 100:.0f}%)"
            )

    fig_size = max(7, min(18, len(class_names) * 0.65))

    plt.figure(figsize=(fig_size, fig_size * 0.85))

    sns.heatmap(
        cm_normalized,
        annot=annotations,
        fmt="",
        cmap="Blues",
        vmin=0,
        vmax=1,
        xticklabels=class_names,
        yticklabels=class_names
    )

    plt.title(f"Normalized Confusion Matrix - {model_name}")
    plt.xlabel("Predicted Identity")
    plt.ylabel("Actual Identity")
    plt.xticks(rotation=45, ha="right")
    plt.yticks(rotation=0)
    plt.tight_layout()
    plt.show()

    return metrics, y_pred, probabilities

from google.colab import files
from PIL import Image
import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf

from tensorflow.keras.applications.mobilenet_v2 import preprocess_input

# 0. Load the matching model trained with augmented features (151 features)
model_fusion_pca_aug = tf.keras.models.load_model("/content/best_fusion_pca_aug.keras")

# Upload satu gambar telinga
uploaded = files.upload()
image_path = next(iter(uploaded.keys()))

# Baca gambar dan sesuaikan ukuran
img = Image.open(image_path).convert("RGB")
img_resized = img.resize(
    (IMG_SIZE, IMG_SIZE),
    Image.Resampling.LANCZOS
)

img_array = np.asarray(img_resized, dtype=np.float32)
img_batch = np.expand_dims(img_array, axis=0)

# A. Preprocessing MobileNetV2
img_mn = preprocess_input(img_batch.copy())

# B. Ekstraksi fitur MobileNetV2 menggunakan feature_model_pca yang sudah ada
mobilenet_features = feature_model_pca.predict(
    img_mn,
    batch_size=1,
    verbose=0
)

# C. Standardisasi menggunakan scaler hasil training
mobilenet_scaled = mobilenet_scaler.transform(
    mobilenet_features
)

# D. Reduksi dimensi menggunakan PCA hasil training
mobilenet_pca_features = mobilenet_pca.transform(
    mobilenet_scaled
)

# E. Ekstraksi fitur GLCM dengan fungsi yang sama saat training
glcm_features = extract_glcm_features(
    np.asarray(img_resized, dtype=np.uint8)
).reshape(1, -1)

# F. Standardisasi GLCM
glcm_scaled = glcm_scaler.transform(
    glcm_features
)

# G. Gabungkan fitur dengan urutan yang sama saat training
combined_features = np.concatenate(
    [mobilenet_pca_features, glcm_scaled],
    axis=1
)

# H. Pastikan dimensi sesuai dengan model yang baru
assert combined_features.shape[1] == model_fusion_pca_aug.input_shape[1], (
    f"Dimensi fitur {combined_features.shape[1]} tidak sesuai "
    f"dengan input model {model_fusion_pca_aug.input_shape[1]}"
)

# I. Prediksi menggunakan model yang sesuai
probabilities = model_fusion_pca_aug.predict(
    combined_features,
    verbose=0
)

predicted_index = int(np.argmax(probabilities[0]))
confidence = float(probabilities[0][predicted_index] * 100)

# J. Tampilkan hasil
plt.figure(figsize=(5, 5))
plt.imshow(img_resized)
plt.axis("off")
plt.title("Gambar Uji Identifikasi Telinga")
plt.show()

print("Prediksi identitas:", class_names[predicted_index])
print(f"Confidence: {confidence:.2f}%")
